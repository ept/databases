% One interword space after a full stop or colon, rather than TeX's default of a wider
% space at the end of a sentence. Set at \begin{document} so that nothing in the preamble's
% language setup can reset it. It also keeps code such as \code{{title: ...}} and the
% labels in diagrams from getting a sentence-ending space after a colon.
\AtBeginDocument{\frenchspacing}
% Without the wider spaces after full stops, a paragraph has less room to stretch, and a
% line ending in an unbreakable \code{...} could otherwise overflow the margin. This extra
% stretch is used only for paragraphs that cannot be set without it.
\setlength{\emergencystretch}{1.5em}
% Pygments' Cypher lexer doesn't know the division operator / or GQL's quantifier -+, and
% marks them as errors, which minted draws in a red box. Since the query in a listing has
% been run, an "error" token is the lexer's mistake, not the query's: draw it as plain
% text. The style definitions are reloaded for every listing, so the override has to be
% applied each time, after them.
\makeatletter
\apptocmd{\minted@defstyle}{\@namedef{PYG@tok@err}{}}{}{\PackageWarning{setup}{Could not patch minted}}
\makeatother

% \code{...} is \texttt for identifiers and code fragments, taking its argument literally:
% write \code{has_genre} rather than \texttt{has\_genre}. Unlike \verb it works inside a
% beamer frame -- \verb there needs \begin{frame}[fragile], because beamer re-reads frame
% bodies to build overlays -- and inside the argument of another command. Braces must still
% balance, and it cannot contain other commands, since the argument is detokenised.
%
% The T1 typewriter font has ligatures for << and >>: they come out as guillemets, so a
% plain \texttt{->>} (DuckDB's JSON operator) is set as "-»". \code therefore puts a zero
% kern after every < and >, which breaks the ligature and changes nothing else, since
% typewriter fonts have no kerning. minted listings are unaffected: fancyvrb turns
% ligatures off itself. \tl_to_str:n is \detokenize, as the definition used before.
% The same goes for -- (an en dash in this font), which Cypher's --> needs. Likewise, ' is
% set as a curly closing quote, which is wrong for SQL string literals, so \code replaces it
% with the straight \textquotesingle (from textcomp).
\ExplSyntaxOn
\tl_new:N \l__code_tl
\cs_new_protected:Npn \code #1
  {
    \tl_set:Nx \l__code_tl { \tl_to_str:n {#1} }
    \tl_replace_all:Nnn \l__code_tl { > } { > \kern 0pt \relax }
    \tl_replace_all:Nnn \l__code_tl { < } { < \kern 0pt \relax }
    \tl_replace_all:Nnn \l__code_tl { - } { - \kern 0pt \relax }
    \tl_replace_all:Nnn \l__code_tl { ' } { \textquotesingle }
    \texttt { \l__code_tl }
  }
\ExplSyntaxOff

% Relational algebra. The outer-join symbols are not in any font we have, so they are built
% from a bowtie with rules either side; measured, they come out 12.2pt wide for the one-sided
% ones and 14.6pt for the two-sided one, against 9.7pt for a plain \bowtie.
\newcommand{\ojoin}{\setbox0=\hbox{$\bowtie$}%
    \rule[-.02ex]{.25em}{.4pt}\llap{\rule[\ht0]{.25em}{.4pt}}}
\newcommand{\louterjoin}{\mathbin{\ojoin\mkern-5.8mu\bowtie}}
\newcommand{\routerjoin}{\mathbin{\bowtie\mkern-5.8mu\ojoin}}
\newcommand{\fouterjoin}{\mathbin{\ojoin\mkern-5.8mu\bowtie\mkern-5.8mu\ojoin}}
% Attribute and relation names in relational algebra expressions, so that the subscripts of
% \sigma and \pi are upright rather than italic strings of single-letter variables.
\newcommand{\ra}[1]{\mathrm{#1}}
% Relational algebra operators are set one size larger than the text around them, so that
% they stand out from the relation names, subscripts and parentheses, which stay at the
% size of the surrounding text. \larger is relative, so this works at \footnotesize on a
% slide as well as at \normalsize in the notes; measured, a sigma goes from 7.11 to 8.53pt
% wide. \mathlarger from the same package does nothing to these symbols, which is why the
% glyph is set in an \mbox instead; the surrounding \mathord or \mathbin puts the maths
% class back, so spacing and subscripts behave as they would otherwise.
\newcommand{\raop}[1]{\mathord{\mbox{\larger[1]$#1$}}}   % sigma, pi, rho
\newcommand{\rabin}[1]{\mathbin{\mbox{\larger[1]$#1$}}}  % join, product, set operators

% Records the position of the point where it is used, so that a later
% tikzpicture with [remember picture,overlay] can draw relative to it -- used to put
% braces beside particular rows of a table. Needs two LaTeX runs, since the position
% travels via the .aux file. A row's baseline is at the mark; with \arraystretch=1.3
% the row extends 10.92pt above it and 4.68pt below (1.3 x 0.7 and 1.3 x 0.3 of the
% 12pt \footnotesize \baselineskip).
\newcommand{\tikzmark}[1]{\tikz[remember picture,overlay]\coordinate (#1);}

% Like \tikzmark, but keeps the extent of the text rather than recording only a point, so
% that a later [remember picture,overlay] tikzpicture can fit= a shape around it -- a circle
% round a value, or a brace under a whole column. examples/render.py emits these from its
% `mark:` directive, naming the header cell <prefix>0 and the data rows <prefix>1 upwards.
% A small cylinder, the conventional picture of a disk or a stored dataset.
\newcommand{\storageSymbol}{\begin{tikzpicture}%
    \draw (0.4, 0.7) ellipse (0.4 and 0.2);
    \draw (0.0, 0.0) arc (180:360:0.4 and 0.2);
    \draw (0.0, 0) -- (0.0, 0.7);
    \draw (0.8, 0) -- (0.8, 0.7);
\end{tikzpicture}}
