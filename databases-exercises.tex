% The exercises without their solutions, for students. Like databases-solutions.tex, it
% must be built after databases-notes.tex, which writes exercises.tex.
%
% In the lecture notes the exercises are broken up by the surrounding prose; here they are
% the whole document, so each one after the first gets extra space above it, on top of the
% frame's own skipabove. exercises.tex emits \exercisesep before each exercise, outside the
% frame that setup.tex puts around it; the counter still holds the previous exercise's
% number at that point, which is how the first one is left without the extra space.
\newcommand{\exercisesep}{\ifnum\value{exercise}>0 \par\vspace{1.5em}\fi}

% Each exercise in exercises.tex is followed by its solution, wrapped in this command;
% here it discards the solution.
\newcommand{\exercisesolution}[1]{}
